//发送单路波形数据
void CXE_DoITDlg::SendArray_Wave_Mult(        int address_ma,//地址码
											  int bao_long,  //包长
											  int zhilingma_B3,//指令码
											  int zhilingma_B4,//指令码
											  int channel_num,//通道数
											  unsigned char waveType,//波形类型
											  unsigned char ffeng[4],//峰峰值
											  unsigned char plv[4],//频率
											  unsigned char pzhi[4])//偏置
{
	CByteArray array; 
	array.RemoveAll();
    
	array.SetSize(bao_long);

	unsigned check;
	array.SetAt(0,0xaa);//起始位

	array.SetAt(1,address_ma); //地址吗
	array.SetAt(2,bao_long);//命令长度
	array.SetAt(3,zhilingma_B3);//指令码1
	array.SetAt(4,zhilingma_B4);//指令码2
	array.SetAt(5,channel_num);//通道
	array.SetAt(6,waveType);//波形类型
	
	for(int i=7;i<11;i++)
	   array.SetAt(i,ffeng[i-7]);
	for(i=11;i<15;i++)
	   array.SetAt(i,plv[i-11]);
	for(i=15;i<19;i++)
	   array.SetAt(i,pzhi[i-15]);
	check=0;
	for(i=0;i<19;i++)
		check=check^array.GetAt(i);
    array.SetAt(19,check);
	//////////////
    m_Serial_Volt.SetOutput(COleVariant(array));
}
